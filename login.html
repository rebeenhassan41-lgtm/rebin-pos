async function login() { const userField = document.getElementById('username'); const passField = document.getElementById('password'); const btn = document.getElementById('loginBtn'); const err = document.getElementById('errorMsg'); const userValue = userField.value.trim(); const passValue = passField.value.trim(); if (!userValue || !passValue) { err.innerText = "تکایە ناو و پاسوۆرد بنووسە!"; err.style.display = "block"; return; } btn.disabled = true; btn.innerText = "خەریکی پشکنینە..."; err.style.display = "none"; try { // گەڕان بەدوای یوسەردا const { data, error } = await _supabase .from('system_users') .select('*') .eq('username', userValue); if (error) { console.error("Supabase Error:", error); throw new Error("کێشە لە پەیوەندی داتابەیس هەیە"); } if (data && data.length > 0) { const dbUser = data[0]; // بەراوردکردنی پاسوۆرد (دڵنیابەوە کە لە داتابەیس ستوونی password هەمان ناوە) if (String(dbUser.password) === String(passValue)) { localStorage.setItem("rebeenAuth", "true"); localStorage.setItem("userName", dbUser.full_name); localStorage.setItem("userPosition", dbUser.position); window.location.replace("dashboard.html"); } else { err.innerText = "پاسوۆردەکە هەڵەیە! ❌"; err.style.display = "block"; } } else { err.innerText = "ئەم ناوە ( " + userValue + " ) بوونی نییە! ❌"; err.style.display = "block"; } } catch (e) { err.innerText = "هەڵە: " + e.message; err.style.display = "block"; } finally { btn.disabled = false; btn.innerText = "چوونە ژوورەوە"; } }
