async function login() { const userV = document.getElementById('username').value.trim(); const passV = document.getElementById('password').value.trim(); const btn = document.getElementById('loginBtn'); const err = document.getElementById('errorMsg'); if (!userV || !passV) return alert("تکایە خانەکان پڕ بکەرەوە"); btn.disabled = true; btn.innerText = "خەریکی پشکنینە..."; err.style.display = "none"; try { // گەڕانی ڕاستەوخۆ بە ناو و پاسوۆرد const { data, error } = await _supabase .from('system_users') .select('*') .eq('username', userV) .eq('password', passV); if (error) throw error; if (data && data.length > 0) { // ئەگەر دۆزرایەوە localStorage.setItem("rebeenAuth", "true"); localStorage.setItem("userName", data[0].full_name); localStorage.setItem("userPosition", data[0].position); window.location.replace("dashboard.html"); } else { // ئەگەر نەدۆزرایەوە err.innerText = "ناو یان پاسوۆرد هەڵەیە! ❌"; err.style.display = "block"; btn.disabled = false; btn.innerText = "چوونە ژوورەوە"; } } catch (e) { alert("هەڵەیەک ڕوویدا: " + e.message); btn.disabled = false; btn.innerText = "چوونە ژوورەوە"; } }
